cmake_minimum_required(VERSION 3.15)
project(cic25_single_hop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
find_package(OpenFHE 1.2.0 EXACT CONFIG REQUIRED)
if(OpenFHE_OPENMP)
  message(FATAL_ERROR "Build the pinned OpenFHE dependency with WITH_OPENMP=OFF")
endif()
if(NOT OpenFHE_NATIVE_SIZE STREQUAL "64" OR NOT OpenFHE_BACKEND STREQUAL "4")
  message(FATAL_ERROR "The comparison requires NATIVE_SIZE=64 and MATHBACKEND=4")
endif()
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${OpenFHE_CXX_FLAGS}")
set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} ${OpenFHE_EXE_LINKER_FLAGS}")
include_directories(${OpenFHE_INCLUDE} ${OpenFHE_INCLUDE}/third-party/include
                    ${OpenFHE_INCLUDE}/core ${OpenFHE_INCLUDE}/pke ${OpenFHE_INCLUDE}/binfhe)
link_directories(${OpenFHE_LIBDIR})
set(CMAKE_BUILD_RPATH "${OpenFHE_LIBDIR}")
add_executable(bench_cic25 bench_cic25.cpp)
target_link_libraries(bench_cic25 PRIVATE ${OpenFHE_SHARED_LIBRARIES})
target_compile_definitions(bench_cic25 PRIVATE
    CIC_OPENFHE_VARIANT="n4096-compact-q54-p55"
    CIC_OPENFHE_VERSION="${BASE_OPENFHE_VERSION}")
target_compile_options(bench_cic25 PRIVATE -Wno-unused-variable)
